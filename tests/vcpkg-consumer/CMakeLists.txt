cmake_minimum_required(VERSION 3.25)
project(vcpkg_consumer LANGUAGES CXX)
file(REAL_PATH "${EXPECTED_PREFIX}" EXPECTED_PREFIX)

find_package(pde_solver_amr_core 0.1.0 EXACT CONFIG REQUIRED)
set(library_target pde_solver_amr_core::core)
get_target_property(library_type ${library_target} TYPE)
if(EXPECTED_LINKAGE STREQUAL "static" AND NOT library_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Expected a static library, got ${library_type}")
elseif(EXPECTED_LINKAGE STREQUAL "dynamic" AND NOT library_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "Expected a shared library, got ${library_type}")
endif()

# Every public include and imported binary must be inside the relocated install.
foreach(property INTERFACE_INCLUDE_DIRECTORIES IMPORTED_LOCATION_DEBUG IMPORTED_LOCATION_RELEASE)
    get_target_property(paths ${library_target} ${property})
    if(NOT paths)
        message(FATAL_ERROR "Missing installed target property: ${property}")
    endif()
    foreach(path IN LISTS paths)
        file(REAL_PATH "${path}" real_path)
        cmake_path(IS_PREFIX EXPECTED_PREFIX "${real_path}" NORMALIZE installed)
        if(NOT installed OR NOT EXISTS "${path}")
            message(FATAL_ERROR "${property} escapes the install: ${path}")
        endif()
        message(STATUS "Installed ${property}: ${path}")
    endforeach()
endforeach()
if(WIN32 AND library_type STREQUAL "SHARED_LIBRARY")
    foreach(config DEBUG RELEASE)
        get_target_property(path ${library_target} IMPORTED_IMPLIB_${config})
        file(REAL_PATH "${path}" real_path)
        cmake_path(IS_PREFIX EXPECTED_PREFIX "${real_path}" NORMALIZE installed)
        if(NOT installed OR NOT EXISTS "${path}")
            message(FATAL_ERROR "Import library escapes the install: ${path}")
        endif()
    endforeach()
endif()
set(config_variable "pde_solver_amr_core_DIR")
get_filename_component(config_path "${${config_variable}}" REALPATH)
cmake_path(IS_PREFIX EXPECTED_PREFIX "${config_path}" NORMALIZE installed)
if(NOT installed)
    message(FATAL_ERROR "Package config escapes the install: ${config_path}")
endif()

# Compile each installed public header in isolation. These files are generated
# entirely from the relocated install while source headers are hidden.
file(GLOB_RECURSE public_headers RELATIVE "${EXPECTED_PREFIX}/include"
    "${EXPECTED_PREFIX}/include/pde_solver_amr_core/*.h")
set(header_probes)
foreach(header IN LISTS public_headers)
    string(MAKE_C_IDENTIFIER "${header}" identifier)
    set(probe "${CMAKE_CURRENT_BINARY_DIR}/header_${identifier}.cpp")
    file(WRITE "${probe}" "#include <${header}>\nint main() {}\n")
    add_executable(header_${identifier} "${probe}")
    target_link_libraries(header_${identifier} PRIVATE ${library_target})
    list(APPEND header_probes ${probe})
endforeach()
if(NOT header_probes)
    message(FATAL_ERROR "No installed public headers found")
endif()
add_executable(consumer main.cpp peer.cpp)
target_link_libraries(consumer PRIVATE ${library_target})
enable_testing()
add_test(NAME consumer COMMAND consumer)
